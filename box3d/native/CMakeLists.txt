cmake_minimum_required(VERSION 3.22)
project(ogb3 VERSION 0.1.0 LANGUAGES C)
include(CTest)
set(OGB3_ENGINE_SOURCE "" CACHE PATH "Verified exact Box3D source (driver supplies this)")
option(OGB3_DISABLE_SIMD "Disable native SIMD" OFF)
option(OGB3_TESTING "Compile controlled snapshot test hooks" ${BUILD_TESTING})
option(OGB3_NATIVE_OBSERVATION "Opt-in diagnostic phase observer (not production ABI)" OFF)
if(NOT EXISTS "${OGB3_ENGINE_SOURCE}/src/world_snapshot.c")
  message(FATAL_ERROR "Use build/build_box3d.sh or supply verified OGB3_ENGINE_SOURCE")
endif()
if(NOT CMAKE_C_COMPILER_ID MATCHES "Clang|GNU")
  message(FATAL_ERROR "Initial targets require Clang or GCC (explicit FP/assert policy)")
endif()
# All engine C sources, matching the pinned upstream src/CMakeLists.txt; no
# upstream C++ world/session, samples, unit-test helpers or optional fetches.
file(GLOB engine_sources CONFIGURE_DEPENDS "${OGB3_ENGINE_SOURCE}/src/*.c")
add_library(ogb3_engine OBJECT ${engine_sources})
set(glue_sources api.c motion.c rollback_shim.c)
if(OGB3_NATIVE_OBSERVATION)
  list(APPEND glue_sources native_observe.c)
endif()
if(EXISTS "${CMAKE_CURRENT_SOURCE_DIR}/owned_snapshot.c")
  list(APPEND glue_sources owned_snapshot.c)
  set(has_snapshots ON)
endif()
if(EXISTS "${CMAKE_CURRENT_SOURCE_DIR}/state_dump.c")
  list(APPEND glue_sources state_dump.c)
  set(has_state_dump ON)
endif()
if(EXISTS "${CMAKE_CURRENT_SOURCE_DIR}/recovery_codec.c")
  list(APPEND glue_sources recovery_codec.c authoring_catalog.c)
  set(has_recovery ON)
endif()
add_library(ogb3_glue OBJECT ${glue_sources})
foreach(target ogb3_engine ogb3_glue)
  set_target_properties(${target} PROPERTIES C_STANDARD 17 C_STANDARD_REQUIRED YES C_EXTENSIONS YES POSITION_INDEPENDENT_CODE YES)
  target_include_directories(${target} PRIVATE "${OGB3_ENGINE_SOURCE}/include" "${OGB3_ENGINE_SOURCE}/src" "${CMAKE_CURRENT_SOURCE_DIR}")
  # Keep assertions AND heavy solver validation in optimized configurations.
  target_compile_definitions(${target} PRIVATE BOX3D_VALIDATE B3_ENABLE_ASSERT)
  target_compile_options(${target} PRIVATE -UNDEBUG -fno-fast-math -ffp-contract=off -Wall -Wextra)
  if(OGB3_DISABLE_SIMD)
    target_compile_definitions(${target} PRIVATE BOX3D_DISABLE_SIMD)
  endif()
endforeach()
set_target_properties(ogb3_engine ogb3_glue PROPERTIES C_VISIBILITY_PRESET hidden)
if(has_snapshots)
  target_compile_definitions(ogb3_glue PRIVATE OGB3_HAS_SNAPSHOTS)
endif()
if(has_state_dump)
  target_compile_definitions(ogb3_glue PRIVATE OGB3_HAS_STATE_DUMP)
endif()
if(has_recovery)
  target_compile_definitions(ogb3_glue PRIVATE OGB3_HAS_RECOVERY)
endif()
if(OGB3_TESTING)
  target_compile_definitions(ogb3_glue PRIVATE OGB3_TESTING=1)
endif()
if(OGB3_NATIVE_OBSERVATION)
  target_compile_definitions(ogb3_glue PRIVATE OGB3_NATIVE_OBSERVATION=1)
endif()
# Embed actual content + effective build inputs, not a caller-provided label.
# Paths are relative content identities; target/compiler/SDK/options are explicit.
if(EXISTS "${CMAKE_CURRENT_SOURCE_DIR}/recovery_codec.c")
  file(GLOB_RECURSE identity_files CONFIGURE_DEPENDS
    "${CMAKE_CURRENT_SOURCE_DIR}/*.c" "${CMAKE_CURRENT_SOURCE_DIR}/*.h"
    "${CMAKE_CURRENT_SOURCE_DIR}/*.patch" "${OGB3_ENGINE_SOURCE}/src/*.c"
    "${OGB3_ENGINE_SOURCE}/src/*.h" "${OGB3_ENGINE_SOURCE}/src/*.inl"
    "${OGB3_ENGINE_SOURCE}/include/*.h")
  list(APPEND identity_files "${CMAKE_CURRENT_SOURCE_DIR}/CMakeLists.txt" "${CMAKE_CURRENT_SOURCE_DIR}/../dependency.json")
  list(SORT identity_files)
  set_property(DIRECTORY APPEND PROPERTY CMAKE_CONFIGURE_DEPENDS ${identity_files})
  set(identity_records)
  set(identity_input "ogb3-native-recovery-build-v1;")
  foreach(path IN LISTS identity_files)
    file(SHA256 "${path}" content_sha)
    file(RELATIVE_PATH native_relative "${CMAKE_CURRENT_SOURCE_DIR}" "${path}")
    file(RELATIVE_PATH engine_relative "${OGB3_ENGINE_SOURCE}" "${path}")
    if(NOT native_relative MATCHES "^[.][.]/")
      set(identity_name "native/${native_relative}")
    elseif(NOT engine_relative MATCHES "^[.][.]/")
      set(identity_name "engine/${engine_relative}")
    else()
      get_filename_component(identity_name "${path}" NAME)
    endif()
    list(APPEND identity_records "${identity_name}:${content_sha}")
  endforeach()
  list(SORT identity_records)
  string(APPEND identity_input "${identity_records};")
  string(TOUPPER "${CMAKE_BUILD_TYPE}" configuration_upper)
  get_target_property(engine_definitions ogb3_engine COMPILE_DEFINITIONS)
  get_target_property(glue_definitions ogb3_glue COMPILE_DEFINITIONS)
  get_target_property(engine_options ogb3_engine COMPILE_OPTIONS)
  get_target_property(glue_options ogb3_glue COMPILE_OPTIONS)
  string(APPEND identity_input "${CMAKE_C_COMPILER_ID};${CMAKE_C_COMPILER_VERSION};${CMAKE_C_COMPILER_TARGET};${CMAKE_SYSTEM_NAME};${CMAKE_SYSTEM_PROCESSOR};${CMAKE_SIZEOF_VOID_P};${CMAKE_OSX_ARCHITECTURES};${CMAKE_OSX_SYSROOT};${CMAKE_OSX_DEPLOYMENT_TARGET};${CMAKE_C_FLAGS};${CMAKE_C_FLAGS_${configuration_upper}};${CMAKE_BUILD_TYPE};${engine_definitions};${glue_definitions};${engine_options};${glue_options};gnu17;PIC;${CMAKE_SHARED_LINKER_FLAGS};${CMAKE_EXE_LINKER_FLAGS}")
  string(SHA256 recovery_build_id "${identity_input}")
  set(target_input "ogb3-native-target-v1;${CMAKE_C_COMPILER_ID};${CMAKE_C_COMPILER_VERSION};${CMAKE_C_COMPILER_TARGET};${CMAKE_SYSTEM_NAME};${CMAKE_SYSTEM_PROCESSOR};${CMAKE_SIZEOF_VOID_P};${CMAKE_OSX_ARCHITECTURES};${CMAKE_OSX_SYSROOT};${CMAKE_OSX_DEPLOYMENT_TARGET};${CMAKE_C_FLAGS};${CMAKE_C_FLAGS_${configuration_upper}};${CMAKE_BUILD_TYPE};${engine_definitions};${glue_definitions};${engine_options};${glue_options};gnu17;PIC;${CMAKE_SHARED_LINKER_FLAGS};${CMAKE_EXE_LINKER_FLAGS}")
  string(SHA256 native_target_id "${target_input}")
  file(WRITE "${CMAKE_CURRENT_BINARY_DIR}/native-domain-identity.txt"
    "build=${recovery_build_id}\ntarget=${native_target_id}\n${target_input}\n")
  target_compile_definitions(ogb3_glue PRIVATE OGB3_RECOVERY_BUILD_ID="${recovery_build_id}" OGB3_NATIVE_TARGET_ID="${native_target_id}")
endif()
# One archive: Odin consumers need no transitive static link discovery.
add_library(ogb3 STATIC $<TARGET_OBJECTS:ogb3_engine> $<TARGET_OBJECTS:ogb3_glue>)
add_library(ogb3_shared SHARED $<TARGET_OBJECTS:ogb3_engine> $<TARGET_OBJECTS:ogb3_glue>)
set_target_properties(ogb3_shared PROPERTIES OUTPUT_NAME ogb3)
foreach(target ogb3 ogb3_shared)
  target_include_directories(${target} PUBLIC "${CMAKE_CURRENT_SOURCE_DIR}")
  if(UNIX AND NOT APPLE)
    target_link_libraries(${target} PUBLIC m pthread)
  endif()
endforeach()
